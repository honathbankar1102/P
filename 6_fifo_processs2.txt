#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <errno.h>

#define FIFO1 "fifo1"
#define FIFO2 "fifo2"
#define BUF 1024

static int make_fifo(const char *name) {
    if (mkfifo(name, 0666) == -1 && errno != EEXIST) {
        perror("mkfifo");
        return -1;
    }
    return 0;
}

int main(void) {
    char buffer[BUF];
    ssize_t n;
    long characters = 0, words = 0, lines = 0;
    int inWord = 0, sawAny = 0, lastWasNewline = 1;

    if (make_fifo(FIFO1) == -1 || make_fifo(FIFO2) == -1) return 1;

    int fd1 = open(FIFO1, O_RDONLY);
    if (fd1 == -1) { perror("open fifo1"); return 1; }

    while ((n = read(fd1, buffer, sizeof(buffer))) > 0) {
        for (ssize_t i = 0; i < n; i++) {
            unsigned char ch = (unsigned char)buffer[i];
            characters++;
            sawAny = 1;
            lastWasNewline = (ch == '\n');

            if (ch == '\n') lines++;

            if (isspace(ch)) {
                inWord = 0;
            } else if (!inWord) {
                words++;
                inWord = 1;
            }
        }
    }
    close(fd1);

    /* Count a final unterminated line as a line too. */
    if (sawAny && !lastWasNewline) lines++;

    FILE *fp = fopen("output.txt", "w");
    if (fp == NULL) { perror("output.txt"); return 1; }
    fprintf(fp, "Number of characters: %ld\n", characters);
    fprintf(fp, "Number of words: %ld\n", words);
    fprintf(fp, "Number of lines: %ld\n", lines);
    fclose(fp);

    fp = fopen("output.txt", "r");
    if (fp == NULL) { perror("output.txt"); return 1; }

    int fd2 = open(FIFO2, O_WRONLY);
    if (fd2 == -1) { perror("open fifo2"); fclose(fp); return 1; }

    while ((n = (ssize_t)fread(buffer, 1, sizeof(buffer), fp)) > 0) {
        ssize_t sent = 0;
        while (sent < n) {
            ssize_t w = write(fd2, buffer + sent, (size_t)(n - sent));
            if (w <= 0) { perror("write fifo2"); fclose(fp); close(fd2); return 1; }
            sent += w;
        }
    }

    fclose(fp);
    close(fd2);
    printf("Analysis completed. Results saved in output.txt\n");
    return 0;
}
