#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <errno.h>

#define FIFO1 "fifo1"
#define FIFO2 "fifo2"
#define BUF 1024

static int make_fifo(const char *name) {
    if (mkfifo(name, 0666) == -1 && errno != EEXIST) {
        perror("mkfifo");
        return -1;
    }
    return 0;
}

int main(void) {
    char buffer[BUF];
    ssize_t n;
    int fd1, fd2;

    if (make_fifo(FIFO1) == -1 || make_fifo(FIFO2) == -1) return 1;

    printf("Enter sentences (type END on a separate line to finish):\n");
    fd1 = open(FIFO1, O_WRONLY);
    if (fd1 == -1) { perror("open fifo1"); return 1; }

    while (fgets(buffer, sizeof(buffer), stdin) != NULL) {
        if (strcmp(buffer, "END\n") == 0 || strcmp(buffer, "END") == 0)
            break;

        size_t len = strlen(buffer), sent = 0;
        while (sent < len) {
            ssize_t w = write(fd1, buffer + sent, len - sent);
            if (w <= 0) { perror("write fifo1"); close(fd1); return 1; }
            sent += (size_t)w;
        }
    }
    close(fd1);

    fd2 = open(FIFO2, O_RDONLY);
    if (fd2 == -1) { perror("open fifo2"); return 1; }

    printf("\nResult received from Process 2:\n");
    while ((n = read(fd2, buffer, sizeof(buffer))) > 0) {
        size_t sent = 0;
        while (sent < (size_t)n) {
            ssize_t w = write(STDOUT_FILENO, buffer + sent, (size_t)n - sent);
            if (w <= 0) { perror("stdout write"); close(fd2); return 1; }
            sent += (size_t)w;
        }
    }
    close(fd2);
    return 0;
}
