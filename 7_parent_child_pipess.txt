#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/types.h>
#include <sys/wait.h>

#define BUF 1024

int main(void) {
    int pipe1[2], pipe2[2];
    pid_t pid;
    char filename[256], buffer[BUF];
    ssize_t n;

    if (pipe(pipe1) == -1 || pipe(pipe2) == -1) {
        perror("pipe");
        return 1;
    }

    pid = fork();
    if (pid < 0) { perror("fork"); return 1; }

    if (pid > 0) {
        /* Parent: send pathname to child and receive file contents. */
        close(pipe1[0]);
        close(pipe2[1]);

        printf("Enter file pathname: ");
        if (fgets(filename, sizeof(filename), stdin) == NULL) {
            fprintf(stderr, "No pathname entered.\n");
            close(pipe1[1]);
            close(pipe2[0]);
            waitpid(pid, NULL, 0);
            return 1;
        }
        filename[strcspn(filename, "\n")] = '\0';

        size_t len = strlen(filename) + 1, sent = 0;
        while (sent < len) {
            ssize_t w = write(pipe1[1], filename + sent, len - sent);
            if (w <= 0) { perror("write pathname"); break; }
            sent += (size_t)w;
        }
        close(pipe1[1]);

        printf("\nFile contents received from child:\n\n");
        while ((n = read(pipe2[0], buffer, sizeof(buffer))) > 0) {
            size_t done = 0;
            while (done < (size_t)n) {
                ssize_t w = write(STDOUT_FILENO, buffer + done, (size_t)n - done);
                if (w <= 0) { perror("write stdout"); break; }
                done += (size_t)w;
            }
        }
        close(pipe2[0]);
        waitpid(pid, NULL, 0);
    } else {
        /* Child: receive pathname, read file, and send its contents. */
        close(pipe1[1]);
        close(pipe2[0]);

        size_t received = 0;
        while (received < sizeof(filename) - 1) {
            n = read(pipe1[0], filename + received,
                     sizeof(filename) - 1 - received);
            if (n <= 0) break;
            received += (size_t)n;
            if (memchr(filename, '\0', received) != NULL) break;
        }
        close(pipe1[0]);

        if (received == 0) {
            fprintf(stderr, "Child: pathname not received.\n");
            close(pipe2[1]);
            _exit(1);
        }
        filename[sizeof(filename) - 1] = '\0';

        int fd = open(filename, O_RDONLY);
        if (fd == -1) {
            perror("Child: open file");
            close(pipe2[1]);
            _exit(1);
        }

        while ((n = read(fd, buffer, sizeof(buffer))) > 0) {
            ssize_t sent = 0;
            while (sent < n) {
                ssize_t w = write(pipe2[1], buffer + sent, (size_t)(n - sent));
                if (w <= 0) { perror("Child: write pipe"); close(fd); close(pipe2[1]); _exit(1); }
                sent += w;
            }
        }
        close(fd);
        close(pipe2[1]);
        _exit(0);
    }
    return 0;
}
