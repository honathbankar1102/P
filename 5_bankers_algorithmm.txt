#include <stdio.h>

#define MAX 10

int main(void) {
    int n, m;
    int allocation[MAX][MAX], max[MAX][MAX], need[MAX][MAX];
    int available[MAX], work[MAX], finish[MAX] = {0};
    int safe[MAX], count = 0;

    printf("Enter number of processes (1-%d): ", MAX);
    if (scanf("%d", &n) != 1 || n < 1 || n > MAX) {
        printf("Invalid number of processes.\n");
        return 1;
    }

    printf("Enter number of resource types (1-%d): ", MAX);
    if (scanf("%d", &m) != 1 || m < 1 || m > MAX) {
        printf("Invalid number of resource types.\n");
        return 1;
    }

    printf("Enter Allocation Matrix:\n");
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (scanf("%d", &allocation[i][j]) != 1 || allocation[i][j] < 0) {
                printf("Allocation values must be non-negative integers.\n");
                return 1;
            }
        }
    }

    printf("Enter Max Matrix:\n");
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (scanf("%d", &max[i][j]) != 1 ||
                max[i][j] < allocation[i][j]) {
                printf("Each Max value must be >= its Allocation value.\n");
                return 1;
            }
            need[i][j] = max[i][j] - allocation[i][j];
        }
    }

    printf("Enter Available Resources:\n");
    for (int j = 0; j < m; j++) {
        if (scanf("%d", &available[j]) != 1 || available[j] < 0) {
            printf("Available resources must be non-negative integers.\n");
            return 1;
        }
        work[j] = available[j];
    }

    printf("\nNeed Matrix:\n");
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) printf("%d ", need[i][j]);
        printf("\n");
    }

    while (count < n) {
        int found = 0;

        for (int i = 0; i < n; i++) {
            if (finish[i] == 0) {
                int j;
                for (j = 0; j < m; j++) {
                    if (need[i][j] > work[j]) break;
                }

                if (j == m) {
                    for (j = 0; j < m; j++)
                        work[j] += allocation[i][j];

                    safe[count++] = i;
                    finish[i] = 1;
                    found = 1;
                }
            }
        }

        if (!found) {
            printf("\nSystem is NOT in a safe state.\n");
            return 0;
        }
    }

    printf("\nSystem is in a safe state.\nSafe Sequence: ");
    for (int i = 0; i < n; i++) printf("P%d ", safe[i]);
    printf("\n");
    return 0;
}
